CREATE TABLE SALES_ORDER_23
(
    ORDERNO VARCHAR2(6) PRIMARY KEY
        CHECK (ORDERNO LIKE 'O%'),

    CLIENTNO VARCHAR2(6),
    
    ORDERDATE DATE NOT NULL,

    DELYADDR VARCHAR2(25),

    SALESMANNO VARCHAR2(6),

    DELYTYPE CHAR(1) DEFAULT 'F'
        CHECK (DELYTYPE IN ('P', 'F')),

    BILLYN CHAR(1),

    DELYDATE DATE
        CHECK (DELYDATE >= ORDERDATE),

    ORDERSTATUS VARCHAR2(10)
        CHECK (ORDERSTATUS IN
        ('IN PROCESS', 'FULFILLED', 'BACKORDER', 'CANCELLED')),

    CONSTRAINT FK_SALES_ORDER_CLIENT
        FOREIGN KEY (CLIENTNO)
        REFERENCES CLIENT_MASTER_23(CLIENTNO),

    CONSTRAINT FK_SALES_ORDER_SALESMAN
        FOREIGN KEY (SALESMANNO)
        REFERENCES SALESMAN_MASTER_23(SALESMANNO)
);
