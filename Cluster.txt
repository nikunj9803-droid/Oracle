CREATE CLUSTER COLLEGE_CLUSTER
(
    COURSE_ID NUMBER(3)
);

CREATE TABLE STUDENT4
(
    STUDENT_ID NUMBER,
    STUDENT_NAME VARCHAR2(30),
    COURSE_ID NUMBER(3)
)
CLUSTER COLLEGE_CLUSTER(COURSE_ID);

CREATE TABLE COURSE
(
    COURSE_ID NUMBER(3),
    COURSE_NAME VARCHAR2(30)
)
CLUSTER COLLEGE_CLUSTER(COURSE_ID);

CREATE INDEX COLLEGE_CLUSTER_IDX
ON CLUSTER COLLEGE_CLUSTER;
