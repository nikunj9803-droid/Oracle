SELECT P.DESCRIPTION, SUM(S.QTYDISP) AS TOTAL_QTY_SOLD
   FROM PRODUCT_MASTER_23 P, SALES_ORDER_DETAILS_23 S
   WHERE P.PRODUCTNO = S.PRODUCTNO
   GROUP BY P.DESCRIPTION;


 SELECT PRODUCTNO,
          SUM(QTYDISP * PRODUCTRATE) AS VALUE_SOLD
   FROM SALES_ORDER_DETAILS_23
   GROUP BY PRODUCTNO;


 SELECT O.CLIENTNO,
          AVG(D.QTYDISP) AS AVG_QTY_SOLD
   FROM SALES_ORDER_23 O, SALES_ORDER_DETAILS_23 D
   WHERE O.ORDERNO = D.ORDERNO
   GROUP BY O.CLIENTNO
   HAVING SUM(D.QTYDISP * D.PRODUCTRATE) = 15000;


 SELECT SUM(D.QTYDISP * D.PRODUCTRATE) AS TOTAL_BILLED_ORDERS
   FROM SALES_ORDER_23 O, SALES_ORDER_DETAILS_23 D
   WHERE O.ORDERNO = D.ORDERNO
   AND O.BILLYN = 'Y'
   AND TO_CHAR(O.ORDERDATE, 'MON') = 'JUN';

SELECT P.PRODUCTNO, P.DESCRIPTION
FROM CLIENT_MASTER_23 C,
     SALES_ORDER_23 O,
     SALES_ORDER_DETAILS_23 D,
     PRODUCT_MASTER_23 P
WHERE C.CLIENTNO = O.CLIENTNO
  AND O.ORDERNO = D.ORDERNO
  AND D.PRODUCTNO = P.PRODUCTNO
  AND C.NAME = 'RAKESH JOSHI';

SELECT P.PRODUCTNO, P.DESCRIPTION,
       D.QTYORDERED - D.QTYDISP AS QTY_TO_DELIVER
FROM PRODUCT_MASTER_23 P,
     SALES_ORDER_DETAILS_23 D,
     SALES_ORDER_23 O
WHERE P.PRODUCTNO = D.PRODUCTNO
  AND D.ORDERNO = O.ORDERNO
  AND TO_CHAR(O.DELV_DATE, 'MM-YYYY')
      = TO_CHAR(SYSDATE, 'MM-YYYY')
  AND D.QTYORDERED > D.QTYDISP;


SELECT P.PRODUCTNO, P.DESCRIPTION
FROM PRODUCT_MASTER_23 P,
     SALES_ORDER_DETAILS_23 D
WHERE P.PRODUCTNO = D.PRODUCTNO
GROUP BY P.PRODUCTNO, P.DESCRIPTION
HAVING COUNT(D.ORDERNO) > 3;
